# -------------------------- 核心：手动实现全局跨域（替代flask-cors） --------------------------
@app.after_request
def add_cors_headers(response):
    """全局跨域响应头：所有接口自动添加跨域配置"""
    # 允许所有域名跨域（开发阶段用*，上线替换为小程序具体域名）
    response.headers['Access-Control-Allow-Origin'] = '*'
    # 允许的请求方法（覆盖小程序常用的GET/POST/OPTIONS）
    response.headers['Access-Control-Allow-Methods'] = 'GET, POST, OPTIONS'
    # 允许的请求头（适配JSON请求、自定义头X-Session-Id）
    response.headers['Access-Control-Allow-Headers'] = 'Content-Type, X-Session-Id'
    # 允许携带Cookie（可选，登录态需要时开启）
    response.headers['Access-Control-Allow-Credentials'] = 'true'
    # 预检请求缓存时间（86400秒=24小时，减少OPTIONS请求）
    response.headers['Access-Control-Max-Age'] = '86400'
    return response

# -------------------------- 核心工具函数（原爬虫逻辑保留） --------------------------
def encode_args(user: str, pwd: str) -> str:
    """生成加密参数encoded"""
    user_encode = base64.b64encode(user.encode())
    pwd_encode = base64.b64encode(pwd.encode())
    encoded = user_encode.decode() + "%%%" + pwd_encode.decode()
    return encoded

def submit_login(session, user: str, pwd: str, captcha: str) -> tuple[bool, str]:
    """提交登录验证，返回(是否成功, 提示信息)"""
    encoded = encode_args(user, pwd)
    data = {
        "userAccount": '',
        "userPassword": '',
        "RANDOMCODE": captcha,
        "encoded": encoded
    }
    login_url = 'http://jw.bkty.top:89/jsxsd/xk/LoginToXk'
    try:
        response = session.post(login_url, data=data, headers={"User-Agent": fake_useragent.UserAgent().random},
                                timeout=5)
        if "用户名或密码错误" in response.text:
            return False, "用户名或密码错误"
        elif "验证码错误" in response.text:
            return False, "验证码错误"
        elif base64.b64decode(encoded.split("%%%")[0]).decode() in response.text:
            return True, "登录成功"
        else:
            return False, "未知错误"
    except requests.exceptions.Timeout:
        return False, "请求超时"
    except Exception as e:
        return False, f"登录异常：{str(e)}"

def get_exam(session, semester_stage: str = '', course_nature: str = '', class_name: str = '',

def format_table(html_content: str) -> list:
    """解析课表HTML为二维列表"""
    soup = BeautifulSoup(html_content, 'html.parser')
    table = soup.find('table', {'class': 'table'})
    if not table:
        return []
    course_table = []
    # 处理表头
    header_row = [cell.get_text(strip=True) for cell in table.find('thead').find_all('th')]
    course_table.append(header_row)
    # 处理内容行
    rows = table.find('tbody').find_all('tr')
    for row in rows:
        course_row = []
        cells = row.find_all('td')
        # 第一列（时间/节次）
        time_text = cells[0].get_text(strip=True).replace('\n', ' ')
        course_row.append(time_text)
        # 其他列（课程+地点）
        for cell in cells[1:]:
            course_p = cell.find('p')
            if course_p:
                course_name = course_p.get_text(strip=True)
                course_info = course_p.get('title', '')
                location = ""
                if "上课地点：" in course_info:
                    location_start = course_info.find("上课地点：") + 5
                    location_end = course_info.find("<br/>", location_start)
                    location = course_info[location_start:location_end] if location_end != -1 else course_info[
                                                                                                   location_start:]
                course_row.append(f"{course_name}: {location}")
            else:
                course_row.append("")
        course_table.append(course_row)
    return course_table

def get_basic_url():
    """
    获取通知公告的7条基本URL
    :return: 基本URL列表
    """
    basic_url = 'https://tj.ustb.edu.cn/jwc/tzgg1/jxyx.htm'
    response = requests.get(basic_url)
    response.encoding = 'utf-8'

    # 解析HTML
    tree = etree.HTML(response.text)

    # 使用Xpath提取7个li标签
    li_elements = tree.xpath('/html/body/div[7]/div[1]/ul/*')

    # 提取title和href并以字典形式保存
    categories = []
    for li in li_elements:
        # 提取title
        title = li.xpath('.//a/text()')[0].strip() if li.xpath('.//a/text()') else ''
        # 提取href
        href = li.xpath('.//a/@href')[0] if li.xpath('.//a/@href') else ''
        # 构建完整的URL
        full_href = basic_url.rsplit('/', 1)[0] + '/' + href if href else ''
        # 添加到分类列表
        categories.append({'title': title, 'href': full_href})
        
    return categories

def format_url(detail_href):
    """
    格式化URL
    :param detail_href: 原始URL
    :return: 格式化后的URL
    """
    # 构建完整的URL
    if detail_href:
        if detail_href.startswith('http'):
            full_detail_href = detail_href
        else:
            # 根据当前分类URL构建完整路径
            full_detail_href = "https://tj.ustb.edu.cn/jwc" + detail_href.strip('..')
    else:
        full_detail_href = ''

    return full_detail_href

def get_detail_info(category_url):
    """
    获取分类下的通知公告详细信息
    :param category_url: 分类URL
    :return: 通知公告详细信息字典
    """
    # 发送请求
    category_response = requests.get(category_url)
    category_response.encoding = 'utf-8'
    
    # 解析HTML
    category_tree = etree.HTML(category_response.text)
    
    # 提取指定Xpath路径下的li标签
    detail_li_elements = category_tree.xpath('/html/body/div[7]/div[2]/div[2]/ul/li')
    
    # 提取每个li标签中的title和href
    detail_items = {}
    for li in detail_li_elements:
        # 提取title
        detail_title = li.xpath('.//a/text()')[0].strip() if li.xpath('.//a/text()') else ''
        # 提取href
        detail_href = li.xpath('.//a/@href')[0] if li.xpath('.//a/@href') else ''
        new_url = format_url(detail_href)
        
        # 添加到详细信息字典
        if detail_title:
            detail_items[detail_title] = new_url

    return detail_items


def get_final_result():
    """

    # 访问每个分类链接，提取详细信息

    return final_result

def get_tr_elements(tr_elements):
    # 存储结果的列表
    results = []
    # 遍历每个tr元素
    for tr in tr_elements:
        # 提取当前tr中的所有td元素
        td_elements = tr.xpath("./td")
        # 遍历每个td元素
        for td in td_elements:
            # 提取a标签
            a_elements = td.xpath("./a")
            for a in a_elements:
                # 提取href
                hrefs = a.xpath("./@href")
                if hrefs:
                    href = "https://tj.ustb.edu.cn/news/"+hrefs[0]
                    # 提取文本值（包括所有子节点的文本）
                    text_values = a.xpath(".//text()")
                    text_value = "".join(text_values).strip() if text_values else ""
                    # 以键值对形式存储
                    results.append({text_value: href})
    return results

def get_college_class_news(class_url):
    """
    获取系部动态指定url页面的文章链接
    :param class_url: 系部动态url
    :return: 文章链接字典
    :key: 文章标题
    :value: 包含文章链接和时间信息的字典
    """
    response = requests.get(class_url)
    response.encoding = 'utf-8'

    # 解析HTML
    html = etree.HTML(response.text)
    
    # 定位目标容器
    container = html.xpath("/html/body/div[4]/div[1]/div[2]/ul")[0]
    
    # 提取所有li元素
    li_elements = container.xpath("./li")
    
    # 存储结果的字典
    article_results = {}
    
    # 遍历每个li元素
    for li in li_elements:
        # 提取h1标签内容作为键
        h1_elements = li.xpath(".//h1")
        if not h1_elements:
            continue
        title = "".join(h1_elements[0].xpath(".//text()")).strip()
        if not title:
            continue
        
        # 提取a标签中的href
        a_elements = li.xpath(".//a")
        href = ""
        if a_elements:
            hrefs = a_elements[0].xpath("./@href")
            if hrefs:
                href = hrefs[0]
                # 处理相对路径
                if not href.startswith('http'):
                    href = "https://tj.ustb.edu.cn/news" + href.strip('..')
   
        # 提取div class="time"的span里的时间信息
        time_elements = li.xpath(".//div[@class='time']/span")
        time = ""
        if time_elements:
            time = "".join(time_elements[0].xpath(".//text()")).strip()
        
        # 构建值字典
        value_dict = {
            "href": href,
            "time": time
        }
        
        # 添加到结果字典
        article_results[title] = value_dict
    
    return article_results

class BaseExtractor:
    """基础提取器类，提供通用功能"""
    
    def __init__(self, source):
        """初始化提取器
        
        Args:
            source: 数据源，远程URL
        """
        self.source = source
        self.data = None
    
    def _get_soup_from_url(self, url):
        """从URL获取BeautifulSoup对象
        
        Args:
            url: 远程URL
            
        Returns:
            BeautifulSoup对象
        """
        # 发送HTTP请求获取内容
        response = requests.get(url, timeout=10)
        response.encoding = 'utf-8'  # 设置编码为UTF-8
        
        # 检查响应状态码
        if response.status_code != 200:
            raise requests.HTTPError(f"请求失败，状态码：{response.status_code}")
        
        # 解析HTML内容
        soup = BeautifulSoup(response.text, 'html.parser')
        
        return soup

    """热门借阅数据提取器类"""
    
    def extract_data(self):
        """提取数据"""
        # 从URL获取内容

        # 提取统计信息
        stats_info = self._extract_stats_info(soup)
        
        # 提取分类列表

class BookSearchExtractor(BaseExtractor):
    """图书检索数据提取器类"""
    
    def extract_data(self):
        """提取数据"""
        # 从URL获取内容
        soup = self._get_soup_from_url(self.source)
        
        # 提取书籍列表
        books = self._extract_books(soup)
        
        # 整合数据
        self.data = {
            '书籍列表': books
        }
        
        return self.data
    
    def _extract_books(self, soup):
        """提取书籍列表"""
        books = []
        
        # 查找目标ol元素
        ol_tag = soup.find('ol', id='search_book_list')
        if ol_tag:
            li_tags = ol_tag.find_all('li', class_='book_list_info')
            
            for i, li in enumerate(li_tags):
                # 提取图书分类
                h3_tag = li.find('h3')
                category = h3_tag.find('span').get_text(strip=True) if h3_tag else ''
                
                # 提取详情链接
                a_tag = h3_tag.find('a') if h3_tag else None
                link = a_tag['href'] if (a_tag and a_tag.has_attr('href')) else ''
                
                # 提取标题
                title = a_tag.get_text(strip=True) if a_tag else ''
                # 处理标题，去掉前面的序号和后面的索书号
                if title:
                    # 去掉前面的序号（如"1."）
                    if '.' in title:
                        title = title.split('.', 1)[1].strip()
                    # 去掉后面的索书号（如" TP311.12/37=3  "）
                    if title and len(title) > 5:
                        # 查找最后一个空格后面的索书号
                        last_space_idx = title.rfind(' ')
                        if last_space_idx > 0:
                            # 检查最后部分是否包含/或=等索书号特征
                            if any(char in title[last_space_idx:] for char in ['/', '=', ' ']):
                                title = title[:last_space_idx].strip()
                
                # 提取馆藏复本和可借复本
                p_tag = li.find('p')
                holdings = 0
                available = 0
                if p_tag:
                    span_tag = p_tag.find('span')
                    if span_tag:
                        span_text = span_tag.get_text(strip=True)
                        # 提取馆藏复本
                        if '馆藏复本：' in span_text:
                            holdings_part = span_text.split('馆藏复本：')[1].split('\n')[0].strip()
                            holdings = int(holdings_part) if holdings_part.isdigit() else 0
                        # 提取可借复本
                        if '可借复本：' in span_text:
                            available_part = span_text.split('可借复本：')[1].strip()
                            available = int(available_part) if available_part.isdigit() else 0
                
                # 提取责任者和出版社
                author = ''
                publisher = ''
                if p_tag:
                    # 获取p标签的所有纯文本内容，自动处理HTML标签
                    p_text = p_tag.get_text()
                    # 按换行符分割
                    lines = p_text.split('\n')
                    # 过滤掉空行和只包含空格的行
                    lines = [line.strip() for line in lines if line.strip()]
                    
                    # 提取责任人和出版社
                    if len(lines) >= 3:
                        author_line = lines[1]
                        # 去掉责任者前面的"可借复本：X "内容
                        if '可借复本：' in author_line:
                            author = author_line.split('可借复本：')[-1].strip()
                            # 如果还包含数字，进一步处理
                            if author and any(char.isdigit() for char in author):
                                # 找到第一个数字后的位置
                                for i, char in enumerate(author):
                                    if char.isdigit():
                                        # 跳过数字和后面的空格
                                        author = author.split(' ', 1)[1].strip() if ' ' in author else author
                                        break
                        else:
                            author = author_line
                        
                        publisher = lines[2]
                        # 去掉出版社后面可能的无关内容
                        if publisher and '馆藏' in publisher:
                            publisher = publisher.split('馆藏')[0].strip()
                
                # 构建书籍字典
                book = {
                    '图书分类': category,
                    '详情链接': link,
                    '标题': title,
                    '馆藏复本': holdings,
                    '可借复本': available,
                    '责任者': author,
                    '出版社': publisher
                }
                books.append(book)
        
        return books

class BookDetailExtractor(BaseExtractor):
    """图书检索详情数据提取器类"""
    
    def extract_data(self):
        """提取数据"""
        # 从URL获取内容
        soup = self._get_soup_from_url(self.source)
        
        # 提取书目信息
        book_info = self._extract_book_info(soup)
        
        # 提取馆藏信息
        holdings = self._extract_holdings(soup)
        
        # 整合数据
        self.data = {
            '书目信息': book_info,
            '馆藏信息': holdings
        }
        
        return self.data
    
    def _extract_book_info(self, soup):
        """提取书目信息"""
        book_info = {}
        
        # 查找书目信息区域
        book_info_div = soup.find('div', id='book_info')
        if book_info_div:
            # 查找所有dl.booklist元素
            dl_tags = book_info_div.find_all('dl', class_='booklist')
            for dl in dl_tags:
                dt_tag = dl.find('dt')
                dd_tag = dl.find('dd')
                if dt_tag and dd_tag:
                    # 提取字段名和字段值
                    field_name = dt_tag.get_text(strip=True)
                    # 去除字段名末尾的冒号
                    field_name = field_name.rstrip(':')
                    # 提取纯文本，去除HTML标签
                    field_value = dd_tag.get_text(strip=True)
                    # 保存到字典
                    if field_name and field_value:
                        # 处理重复字段（如个人责任者、中图法分类号）
                        if field_name in book_info:
                            # 如果已有该字段，合并值
                            existing_value = book_info[field_name]
                            if field_value not in existing_value:
                                book_info[field_name] = f"{existing_value}; {field_value}"
                        else:
                            book_info[field_name] = field_value
        
        # 移除空值或无效字段
        if '豆瓣简介：' in book_info and not book_info['豆瓣简介：'].strip():
            del book_info['豆瓣简介：']
        
        return book_info
    
    def _extract_holdings(self, soup):
        """提取馆藏信息"""
        holdings = []
        
        # 查找馆藏信息表格
        item_table = soup.find('table', id='item')
        if item_table:
            # 获取表头
            header_tr = item_table.find('tr', class_='greytext1')
            headers = []
            if header_tr:
                # 提取表头文本
                header_tds = header_tr.find_all('td')
                headers = [td.get_text(strip=True) for td in header_tds]
            
            # 获取数据行
            data_trs = item_table.find_all('tr', class_='whitetext')
            for tr in data_trs:
                data_tds = tr.find_all('td')
                if data_tds and len(data_tds) >= len(headers):
                    holding_item = {}
                    for i, header in enumerate(headers):
                        # 提取单元格文本，处理馆藏地字段
                        if i == 3:  # 馆藏地字段
                            # 提取纯文本，去除HTML标签
                            cell_text = data_tds[i].get_text(strip=True)
                            # 清理文本，去除多余空格
                            cell_text = ' '.join(cell_text.split())
                        else:
                            cell_text = data_tds[i].get_text(strip=True)
                        holding_item[header] = cell_text
                    holdings.append(holding_item)
        
        return holdings

class AlumniHomeExtractor(BaseExtractor):
    """校友之家数据提取器类"""
    
    def extract_data(self):
        """提取校友之家数据"""
        # 使用lxml解析HTML
        response = requests.get(self.source, headers={'User-Agent': fake_useragent.UserAgent().random})
        response.encoding = 'utf-8'
        html = etree.HTML(response.text)
        
        # 目标XPath路径 - 从校友之家.py文件中获取
        xpath = "/html/body/section[2]/div/div/div[2]/div/div[2]/div[1]/ul"
        ul_elements = html.xpath(xpath)
        
        if not ul_elements:
            self.data = {"error": "未找到目标元素"}
            return self.data
        
        results = []
        
        # 处理找到的ul元素
        for ul in ul_elements:
            # 提取所有li子元素
            li_elements = ul.xpath('./li')
            
            # 遍历每个li元素
            for li in li_elements:
                # 尝试使用更通用的XPath查找a标签
                a_tags = li.xpath('.//a')
                
                for a_tag in a_tags:
                    # 提取标题和日期
                    title_text = ''.join(a_tag.xpath('.//text()')).strip()
                    title_parts = title_text.split('\r\n')
                    
                    if len(title_parts) >= 2:
                        title = title_parts[0].strip()
                        date = title_parts[1].strip()
                    else:
                        title = title_text.strip()
                        date = ""
                    
                    # 提取链接
                    href = a_tag.get('href', '')
                    
                    # 处理相对链接
                    if href and not href.startswith('http'):
                        if href.startswith('//'):
                            full_href = f"https:{href}"
                        elif href.startswith('/'):
                            full_href = f"https://tj.ustb.edu.cn{href}"
                        else:
                            # 对于相对路径，根据当前URL构建完整路径
                            if self.source.endswith('/'):
                                base_url = self.source
                            else:
                                base_url = '/'.join(self.source.split('/')[:-1]) + '/'
                            full_href = base_url + href.lstrip('./')
                    else:
                        full_href = href
                    
                    # 检查是否有有效的标题和链接
                    if title and full_href:
                        results.append({
                            '标题': title,
                            '日期': date,
                            '链接': full_href.replace("index/", "")
                        })
        
        # 去重
        seen = set()
        unique_results = []
        for item in results:
            if item['链接'] not in seen:
                seen.add(item['链接'])
                unique_results.append(item)
        
        self.data = {
            'items': unique_results
        }
        
        return self.data

class ResearchProjectExtractor(BaseExtractor):
    """科研项目数据提取器类"""
    import re
    def __init__(self, source):
        """初始化提取器
        
        Args:
            source: 数据源URL，默认使用科研项目首页
        """
        # 如果没有提供source，使用默认URL
        if not source:
            source = "https://tj.ustb.edu.cn/index/xsky.htm"
        super().__init__(source)
    
    def extract_data(self):
        """提取科研项目数据"""
        try:
            # 获取总页数
            total_pages = self._get_total_pages()
            
            all_projects = []
            
            # 首先访问首页
            index_projects = self._extract_research_projects(self.source)
            all_projects.extend(index_projects)
            
            # 然后从总页数-1开始倒叙访问
            start_page = total_pages - 1
            if start_page > 0:
                base_url = "https://tj.ustb.edu.cn"
                for page in range(start_page, 0, -1):
                    page_url = f"{base_url}/index/xsky/{page}.htm"
                    page_projects = self._extract_research_projects(page_url)
                    all_projects.extend(page_projects)
            
            # 去重
            seen = set()
            unique_projects = []
            for project in all_projects:
                if project['链接'] not in seen:
                    seen.add(project['链接'])
                    unique_projects.append(project)
            
            self.data = {
                'total': len(unique_projects),
                'items': unique_projects
            }
            
        except Exception as e:
            self.data = {
                'error': f'提取数据时出错: {str(e)}'
            }
        
        return self.data
    
    def _get_total_pages(self):
        """获取总页数
        
        Returns:
            int: 总页数
        """
        try:
            # 发送请求获取页面内容
            response = requests.get(self.source, headers={'User-Agent': fake_useragent.UserAgent().random})
            response.encoding = 'utf-8'
            
            # 解析HTML
            html = etree.HTML(response.text)
            
            # 目标XPath
            nums_xpath = "/html/body/div[4]/div/div/div/span[2]/span[9]/a"
            
            # 尝试使用主XPath
            page_elements = html.xpath(nums_xpath)
            
            # 备选XPath
            alternative_xpaths = [
                "//div[@class='pagination']//a[last()]",
                "//span[contains(@class, 'page')]//a[last()]",
                "//*[contains(text(), '页')]//a[last()]"
            ]
            
            # 如果主XPath失败，尝试备选XPath
            if not page_elements:
                for alt_xpath in alternative_xpaths:
                    page_elements = html.xpath(alt_xpath)
                    if page_elements:
                        break
            
            if page_elements:
                # 提取总页数
                page_element = page_elements[0]
                page_text = page_element.text.strip() if page_element.text else ""
                # 提取数字
                total_pages = int(''.join(filter(str.isdigit, page_text)))
                return total_pages
            else:
                return 1
                
        except Exception as e:
            return 1
    
    def _extract_research_projects(self, url):
        """从指定URL提取科研项目信息
        
        Args:
            url: 页面URL
            
        Returns:
            list: 科研项目信息列表
        """
        try:
            # 发送请求获取页面内容
            response = requests.get(url, headers={'User-Agent': fake_useragent.UserAgent().random})
            response.encoding = 'utf-8'
            
            # 解析HTML
            html = etree.HTML(response.text)
            
            # 目标XPath
            data_xpath = "/html/body/div[4]/div/ul"
            
            # 提取目标ul元素
            ul_elements = html.xpath(data_xpath)
            
            if not ul_elements:
                return []
            
            results = []
            
            # 处理找到的ul元素
            for ul in ul_elements:
                # 提取所有li子元素
                li_elements = ul.xpath('./li')
                
                # 遍历每个li元素
                for li in li_elements:
                    # 提取a标签
                    a_tag = li.find('.//a')
                    if a_tag is not None:
                        # 提取标题
                        title = a_tag.text.strip() if a_tag.text else ''
                        # 尝试获取所有文本内容
                        if not title:
                            title = ''.join(a_tag.xpath('.//text()')).strip()
                        
                        # 提取链接
                        href = a_tag.get('href', '')
                        # 处理相对链接
                        if href and not href.startswith('http'):
                            if href.startswith('//'):
                                full_href = f"https:{href}"
                            elif href.startswith('/'):
                                full_href = f"https://tj.ustb.edu.cn{href}"
                            else:
                                full_href = f"https://tj.ustb.edu.cn/{href}"
                        else:
                            full_href = href
                        
                        # 提取日期
                        date = ""
                        # 尝试从li元素中提取日期
                        date_elements = li.xpath('.//*[contains(text(), "-") and contains(text(), "-")]')
                        if date_elements:
                            date = date_elements[0].text.strip() if date_elements[0].text else ""
                        # 尝试从li元素的文本中提取日期
                        if not date:
                            li_text = ''.join(li.xpath('.//text()')).strip()
                            # 查找日期格式的文本
                            import re
                            date_match = re.search(r'\d{4}-\d{2}-\d{2}', li_text)
                            if date_match:
                                date = date_match.group()
                        
                        # 检查是否有有效的标题和链接
                        if title and full_href:
                            results.append({
                                '标题': re.findall(r'\r\n(.*?)\r\n', title)[0],
                                '链接': full_href,
                                '日期': date
                            })
            
            return results
            
        except Exception as e:
            return []

@app.route('/api/ocr/captcha', methods=['POST'])
def api_ocr_captcha():
    """验证码识别API接口
    
    POST参数:
        image: 必需，base64编码的验证码图片数据
    
    返回:
        JSON格式的验证码识别结果
    """
    try:
        # 获取请求参数
        data = request.get_json()
        if not data or 'image' not in data:
            return jsonify({
                'code': 400,
                'message': '缺少image参数',
                'data': None
            })
        
        # 解码base64图片数据
        image_data = data['image']
        # 移除base64前缀（如果有）
        if image_data.startswith('data:image'):
            image_data = image_data.split(',')[1]
        
        # 解码base64数据
        captcha_png = base64.b64decode(image_data)
        
        # 调用ocr_captcha函数识别验证码
        text = ocr_captcha(captcha_png)
        
        # 返回JSON响应
        return jsonify({
            'code': 200,
            'message': 'success',
            'data': {
                'text': text
            }
        })
        
    except Exception as e:
        # 处理异常
        return jsonify({
            'code': 500,
            'message': f'识别验证码失败: {str(e)}',
            'data': None
        })
